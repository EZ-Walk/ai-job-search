%% Compact Arial CV (letter, 1 page)
%% Recreated from the dense single-column layout used in
%% Ruben Aguilar's OpenAI Strategy & Operations Launch Programs resume
%% (ReportLab / Arial). Profile-agnostic skeleton for /apply.
%% Compile with: lualatex -interaction=nonstopmode <file>.tex
%% Output: cv/main_<company>_<role>.tex
%%
%% Packages: article + fontspec, geometry, hyperref, setspace.
%% No enumitem/titlesec (not in TeX Live basic).

\documentclass[10pt,letterpaper]{article}

\usepackage[margin=0.5in]{geometry}
\usepackage{fontspec}
\usepackage{setspace}
\usepackage{hyperref}

\IfFontExistsTF{Arial}{
  \setmainfont{Arial}[
    Ligatures=TeX,
    BoldFont={Arial Bold},
    ItalicFont={Arial Italic},
    BoldItalicFont={Arial Bold Italic}
  ]
}{
  \setmainfont{TeX Gyre Heros}
}

\hypersetup{
  colorlinks=true,
  urlcolor=black,
  linkcolor=black,
  pdftitle={[YOUR_NAME] - CV},
}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\setstretch{1.0}

% Tight hyphen bullets without enumitem
\newenvironment{cvitems}{%
  \begin{list}{-}{%
    \setlength{\leftmargin}{12pt}%
    \setlength{\itemsep}{1.2pt}%
    \setlength{\parsep}{0pt}%
    \setlength{\topsep}{1.5pt}%
    \setlength{\partopsep}{0pt}%
    \setlength{\labelwidth}{6pt}%
    \setlength{\labelsep}{6pt}%
    \setlength{\itemindent}{0pt}%
  }%
}{\end{list}}

% Section: ALL CAPS bold + full-width rule
\newcommand{\cvsection}[1]{%
  \vspace{7pt}%
  \noindent{\bfseries\fontsize{11pt}{13pt}\selectfont\MakeUppercase{#1}}\par
  \vspace{1pt}%
  \noindent\rule{\textwidth}{0.4pt}\par
  \vspace{4pt}%
}

% Company (bold) + location (right); title (italic) + dates (right)
\newcommand{\cventryhead}[4]{%
  \noindent\textbf{#1}\hfill#2\par
  \vspace{0.5pt}%
  \noindent\textit{#3}\hfill#4\par
}

\newcommand{\cveduhead}[4]{%
  \noindent\textbf{#1}\hfill#2\par
  \vspace{0.5pt}%
  \noindent\textit{#3}\hfill#4\par
}
